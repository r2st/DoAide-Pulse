# ---- Auth ----
# JWTs are PyJWT's, not python-jose's. Worth stating because `python-jose` was
# the dependency here once, and removing a line from this file does not remove
# the package from a virtualenv that already has it: `pip install -r` installs
# and upgrades, it never uninstalls.
#
# So both python-jose and its `ecdsa` dependency outlived the requirement. As of
# 2026-08-15 the production box still carries an orphaned `ecdsa` 0.19.2 —
# nothing declares it, nothing imports it, and pip-audit reports it every run
# for PYSEC-2026-1325 (a Minerva timing attack on P-256, no fix released).
#
# Nothing loads it, so it is dead weight rather than exposure, but it cannot be
# cleared from here — only on the box, and only by hand:
#
#   sudo -u herald /opt/Herald/.venv/bin/pip uninstall -y ecdsa python-jose
#
# Until somebody runs that, an `ecdsa` line in a pip-audit report is this, and
# not a dependency that came back.
PyJWT>=2.8
bcrypt>=4.1
# Rate limiting for /auth/* (login guessing, signup, reset-mail flooding).
slowapi>=0.1.9
# Credential encryption at rest (Fernet).
# Floor is 50.0 for PYSEC-2026-3552 (a decryption oracle in the PKCS7 helpers).
# Herald only uses Fernet and never touches those, so the exposure was nil — but
# the floor is what `pip install -r` on the box compares against, and at >=42.0
# an already-installed 49.0.0 satisfied it and was never upgraded.
cryptography>=50.0
